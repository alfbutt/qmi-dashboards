Changes you make here are saved locally for now. To sync them directly to Google Sheets, add this function to your Apps Script and redeploy: 
function doPost(e) {
  try {
    const p = JSON.parse(e.postData.contents);
    SpreadsheetApp.openById(p.id)
      .getSheetByName(p.sheet)
      .getRange(p.row, p.col)
      .setValue(p.value);
    return ContentService
      .createTextOutput(JSON.stringify({ok:true}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({ok:false,e:err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
Then go to Extensions → Apps Script → Deploy → Manage deployments → Edit → New version → Deploy.
